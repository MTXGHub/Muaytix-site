<script>

(function () {
  var root = document.querySelector('.mtx-kp[data-night]');
  if (!root) return;

  var NIGHT = root.getAttribute('data-night');            // 2026-09-21
  var BELL  = (root.getAttribute('data-bell') || '19:00').split(':');
  var FROM  = (root.getAttribute('data-countdown-from') || '15:00').split(':');
  if (!/^\d{4}-\d{2}-\d{2}$/.test(NIGHT)) return;         // bad date, stay quiet

  function two(n) { return (n < 10 ? '0' : '') + n; }

  function clock(h, m) {
    var ampm = h < 12 ? 'AM' : 'PM';
    var h12 = h % 12; if (h12 === 0) h12 = 12;
    return h12 + ':' + two(m) + ' ' + ampm;
  }

  function bangkokNow() {
    var now = new Date();
    return new Date(now.getTime() + (now.getTimezoneOffset() * 60000) + 7 * 3600000);
  }

  function tick() {
    var bkk = bangkokNow();
    var today = bkk.getFullYear() + '-' + two(bkk.getMonth() + 1) + '-' + two(bkk.getDate());

    var state = today < NIGHT ? 'before' : (today > NIGHT ? 'after' : 'tonight');
    root.setAttribute('data-state', state);
    if (state !== 'tonight') return;

    var bellH = parseInt(BELL[0], 10), bellM = parseInt(BELL[1], 10);
    var bell = new Date(bkk.getFullYear(), bkk.getMonth(), bkk.getDate(),
                        bellH, bellM, 0, 0);
    var from = new Date(bkk.getFullYear(), bkk.getMonth(), bkk.getDate(),
                        parseInt(FROM[0], 10), parseInt(FROM[1], 10), 0, 0);
    var left = bell - bkk;
    var say;

    if (left <= 0) {
      say = 'Fighting now — gates stay open all evening';
    } else if (bkk < from) {
      say = 'First bell tonight at ' + clock(bellH, bellM);
    } else {
      var mins = Math.floor(left / 60000);
      var hrs  = Math.floor(mins / 60);
      mins = mins % 60;
      say = 'First bell in ' + (hrs > 0 ? hrs + 'h ' + two(mins) + 'm' :
            mins + (mins === 1 ? ' minute' : ' minutes'));
    }

    var out = root.querySelectorAll('[data-mtx-countdown]');
    for (var i = 0; i < out.length; i++) out[i].textContent = say;
  }

  tick();
  setInterval(tick, 60000);
  document.addEventListener('visibilitychange', function () {
    if (!document.hidden) tick();
  });
})();
</script>

<script>

(function () {
  var sliders = document.querySelectorAll('.mtx-kp [data-mtx-slider]');
  for (var s = 0; s < sliders.length; s++) build(sliders[s]);

  function build(box) {
    var track = box.querySelector('.mtx-kp__shots-track');
    var dots  = box.querySelector('.mtx-kp__shots-dots');
    if (!track || !dots) return;
    var shots = track.querySelectorAll('img');
    if (shots.length < 2) return;   // one photo needs no dots

    for (var i = 0; i < shots.length; i++) {
      var b = document.createElement('button');
      b.type = 'button';
      b.setAttribute('aria-label', 'Photo ' + (i + 1) + ' of ' + shots.length);
      if (i === 0) b.className = 'is-on';
      b.addEventListener('click', go(i));
      dots.appendChild(b);
    }

    function go(i) {
      return function () {
        track.scrollTo({ left: shots[i].offsetLeft - track.offsetLeft, behavior: 'smooth' });
      };
    }

    var waiting = false;
    track.addEventListener('scroll', function () {
      if (waiting) return;
      waiting = true;
      window.requestAnimationFrame(function () {
        waiting = false;
        var at = Math.round(track.scrollLeft / track.clientWidth);
        var buttons = dots.children;
        for (var j = 0; j < buttons.length; j++) {
          buttons[j].className = (j === at) ? 'is-on' : '';
        }
      });
    }, { passive: true });
  }
})();
</script>
