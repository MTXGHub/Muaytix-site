  function hrefOf(card) {
    var a = card && card.querySelector('.mtx-hp__nightcta a');
    return a && a.getAttribute('href');
  }

  var next = live[0];
  var heroBtn = document.querySelector('[data-mtx-tonight]');
  if (heroBtn && hrefOf(next)) {
    heroBtn.setAttribute('href', hrefOf(next));
    heroBtn.textContent =
      next.getAttribute('data-mtx-date') === today ? 'Tonight' : 'Next fight night';
  }

  var satBtn = document.querySelector('[data-mtx-saturday]');
  if (satBtn) {
    for (var m = 0; m < live.length; m++) {
      var href = hrefOf(live[m]);
      if (href && href.indexOf('/rws/') === 0) { satBtn.setAttribute('href', href); break; }
    }
  }
})();
